# fe_tokenizer (owner B): byte-level BPE + pre-tokenizer + added tokens + fast-tokenize chunking + DecodeStream.
add_library(fe_tokenizer STATIC tokenizer.cpp decode_stream.cpp)
target_include_directories(fe_tokenizer PUBLIC ${CMAKE_SOURCE_DIR}/src ${CMAKE_SOURCE_DIR}/third_party/install/include)
target_link_libraries(fe_tokenizer PUBLIC ${CMAKE_SOURCE_DIR}/third_party/install/lib/libonig.a pthread)
target_compile_options(fe_tokenizer PRIVATE -Wall -Wextra)

# tests (tests/tokenizer); golden data from tools/golden/tokenizer_dump.py (tools/golden/tokenizer_data, private)
set(FE_TOK_TESTS ${CMAKE_SOURCE_DIR}/tests/tokenizer)
if(EXISTS ${FE_TOK_TESTS}/tok_unit.cpp)
  add_executable(tok_unit ${FE_TOK_TESTS}/tok_unit.cpp)
  target_link_libraries(tok_unit PRIVATE fe_tokenizer)
  add_test(NAME tokenizer_unit COMMAND tok_unit)
endif()
if(EXISTS ${FE_TOK_TESTS}/tok_golden.cpp)
  add_executable(tok_golden ${FE_TOK_TESTS}/tok_golden.cpp)
  target_link_libraries(tok_golden PRIVATE fe_tokenizer)
  target_compile_definitions(tok_golden PRIVATE FE_SOURCE_DIR="${CMAKE_SOURCE_DIR}")
  add_test(NAME tokenizer_golden COMMAND tok_golden)
  set_tests_properties(tokenizer_golden PROPERTIES SKIP_RETURN_CODE 77)
endif()
if(EXISTS ${FE_TOK_TESTS}/tok_bench.cpp)
  add_executable(tok_bench ${FE_TOK_TESTS}/tok_bench.cpp)
  target_link_libraries(tok_bench PRIVATE fe_tokenizer)
  target_compile_definitions(tok_bench PRIVATE FE_SOURCE_DIR="${CMAKE_SOURCE_DIR}")
endif()
